# Backend D1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Backend Cloudflare Worker + D1 untuk Bayt Mustaqbal app — 15 tabel, semua modul, auth JWT, cron, notifikasi WA.

**Architecture:** Hono + Cloudflare Workers + D1 (SQLite). Subdomain `api.baytmustaqbal.my.id`. Auth JWT HS256 + bcryptjs, HttpOnly cookie. CORS allow `app.baytmustaqbal.my.id`.

**Tech Stack:** TypeScript, Hono, Cloudflare Workers, D1, bcryptjs, jsonwebtoken, Fonnte API

**Spec:** `docs/superpowers/specs/2026-10-19-backend-d1-design.md`

## Global Constraints

- TypeScript, Hono, D1
- Soft delete: kolom `void` (timestamp, NULL = aktif) — **semua DELETE harus set void**
- Rev: kolom `rev` (integer, auto-increment) — **semua UPDATE harus increment rev**
- Query pakai nama kolom (bukan indeks)
- Secrets via `wrangler secret put` — **jangan commit secret**
- CORS: allow `https://app.baytmustaqbal.my.id`, credentials: true
- Response format: `{ ok: true, data: {}, error: null }` / `{ ok: false, data: null, error: { code, message } }`
- Error codes: `UNAUTHENTICATED`, `FORBIDDEN`, `VALIDATION`, `NOT_FOUND`, `CONFLICT`, `SERVER`
- Auth: JWT HS256 + bcryptjs, HttpOnly cookie (`Secure`, `SameSite=Lax`, `Max-Age=7d`)

## Review Focus

- **Soft delete** — semua DELETE harus set `void`, bukan hard delete
- **Rev** — update harus increment `rev` untuk deteksi konflik
- **Auth** — semua `/api/*` (kecuali `/api/auth/login|register`) harus verify JWT
- **CORS** — harus allow `app.baytmustaqbal.my.id` + credentials
- **Validation** — semua input harus divalidasi (required fields, types)

```typescript
import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';

export async function hashPassword(password: string): Promise<string> {
  return bcrypt.hash(password, 12);
}

export async function verifyPassword(password: string, hash: string): Promise<boolean> {
  return bcrypt.compare(password, hash);
}

export function signToken(payload: { userId: string; role: string }, secret: string): string {
  return jwt.sign(payload, secret, { expiresIn: '7d' });
}

export function verifyToken(token: string, secret: string): { userId: string; role: string } | null {
  try {
    return jwt.verify(token, secret) as { userId: string; role: string };
  } catch {
    return null;
  }
}
```

- [ ] **Step 3: Create `middleware/auth.ts`**

```typescript
import { Context, Next } from 'hono';
import { verifyToken } from '../utils/auth';

export async function authMiddleware(c: Context, next: Next) {
  const token = c.req.header('Authorization')?.replace('Bearer ', '') ||
    c.req.cookie('token');
  if (!token) {
    return c.json({ ok: false, data: null, error: { code: 'UNAUTHENTICATED', message: 'Token tidak ada' } }, 401);
  }
  const payload = verifyToken(token, c.env.JWT_SECRET);
  if (!payload) {
    return c.json({ ok: false, data: null, error: { code: 'UNAUTHENTICATED', message: 'Token invalid' } }, 401);
  }
  c.set('userId', payload.userId);
  c.set('role', payload.role);
  await next();
}
```

- [ ] **Step 4: Create `routes/auth.ts`**

```typescript
import { Hono } from 'hono';
import { hashPassword, verifyPassword, signToken } from '../utils/auth';

export const authRoutes = new Hono<{ Bindings: Env }>();

// POST /api/auth/register
authRoutes.post('/register', async (c) => {
  const { username, password, nama } = await c.req.json();
  if (!username || !password || !nama) {
    return c.json({ ok: false, data: null, error: { code: 'VALIDATION', message: 'Field wajib: username, password, nama' } }, 400);
  }
  const hash = await hashPassword(password);
  const id = crypto.randomUUID();
  const now = new Date().toISOString();
  try {
    await c.env.DB.prepare('INSERT INTO users (id, username, password_hash, nama, role, created_at, updated_at) VALUES (?, ?, ?, ?, ?, ?, ?)')
      .bind(id, username, hash, nama, 'user', now, now).run();
    return c.json({ ok: true, data: { id, username, nama, role: 'user' }, error: null });
  } catch (e: any) {
    if (e.message.includes('UNIQUE')) {
      return c.json({ ok: false, data: null, error: { code: 'CONFLICT', message: 'Username sudah ada' } }, 409);
    }
    throw e;
  }
});

// POST /api/auth/login
authRoutes.post('/login', async (c) => {
  const { username, password } = await c.req.json();
  if (!username || !password) {
    return c.json({ ok: false, data: null, error: { code: 'VALIDATION', message: 'Field wajib: username, password' } }, 400);
  }
  const user = await c.env.DB.prepare('SELECT * FROM users WHERE username = ? AND void IS NULL')
    .bind(username).first();
  if (!user) {
    return c.json({ ok: false, data: null, error: { code: 'UNAUTHENTICATED', message: 'Username atau password salah' } }, 401);
  }
  const valid = await verifyPassword(password, user.password_hash);
  if (!valid) {
    return c.json({ ok: false, data: null, error: { code: 'UNAUTHENTICATED', message: 'Username atau password salah' } }, 401);
  }
  const token = signToken({ userId: user.id, role: user.role }, c.env.JWT_SECRET);
  c.header('Set-Cookie', `token=${token}; HttpOnly; Secure; SameSite=Lax; Max-Age=604800; Path=/`);
  return c.json({ ok: true, data: { token, user: { id: user.id, username: user.username, nama: user.nama, role: user.role } }, error: null });
});

// POST /api/auth/logout
authRoutes.post('/logout', (c) => {
  c.header('Set-Cookie', 'token=; HttpOnly; Secure; SameSite=Lax; Max-Age=0; Path=/');
  return c.json({ ok: true, data: null, error: null });
});

// GET /api/auth/me
authRoutes.get('/me', async (c) => {
  const userId = c.get('userId');
  const user = await c.env.DB.prepare('SELECT id, username, nama, avatar_url, role FROM users WHERE id = ? AND void IS NULL')
    .bind(userId).first();
  if (!user) {
    return c.json({ ok: false, data: null, error: { code: 'NOT_FOUND', message: 'User tidak ditemukan' } }, 404);
  }
  return c.json({ ok: true, data: user, error: null });
});
```
